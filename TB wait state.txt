`timescale 1ns / 1ps

module tb_apb_master;
    reg clk;
    reg reset_n;
    reg [31:0] addr;
    reg [31:0] wdata;
    reg EWRITE;
    reg EVALID;
    reg [31:0] PRDATA;
    reg PREADY;
    reg PSLVERR;
    wire [31:0] rdata;
    wire EPREADY;
    wire [31:0] PADDR;
    wire [31:0] PWDATA;
    wire PWRITE;
    wire PSEL;
    wire PENABLE;
    integer wait_states;
    integer i;
    apb_master uut (.clk(clk),.reset_n(reset_n),.addr(addr),.wdata(wdata),.EWRITE(EWRITE),.EVALID(EVALID),.rdata(rdata),.EPREADY(EPREADY),.PRDATA(PRDATA),.PREADY(PREADY),.PSLVERR(PSLVERR),.PADDR(PADDR),.PWDATA(PWDATA),.PWRITE(PWRITE),.PSEL(PSEL),
.PENABLE(PENABLE));
    always #10 clk = ~clk;
  initial
    begin
        clk     = 0;
        reset_n = 0;
        addr    = 0;
        wdata   = 0;
        EWRITE  = 0;
        EVALID  = 0;
        PRDATA  = 0;
        PREADY  = 0;
        PSLVERR = 0;
        repeat(2) @(posedge clk); 
        #1;                       
        reset_n = 1; 
        repeat(3) @(posedge clk); 
        wait_states = $urandom_range(0, 4);
        @(posedge clk);          
        addr   <= 32'hA000_1000;
        wdata  <= 32'hDEAD_DEAF;
        EWRITE <= 1; 
        EVALID <= 1; 

        @(posedge clk);
        EVALID <= 0; 

        for (i = 0; i < wait_states; i = i + 1)
          begin
            @(posedge clk);
            PREADY <= 1'b0; 
        end
        @(posedge clk);
        PREADY <= 1'b1; 

        @(posedge clk);
        PREADY <= 1'b0; 
        addr   <= 32'h0;
        wdata  <= 32'h0;
        EWRITE <= 0; 
        
        #40;

        wait_states = $urandom_range(0, 4);
       
        
        @(posedge clk);
        addr   <= 32'hB000_2000;
        EWRITE <= 0; 
        EVALID <= 1; 

        @(posedge clk);
        EVALID <= 0; 

        for (i = 0; i < wait_states; i = i + 1) begin
            @(posedge clk);
            PREADY <= 1'b0;
            PRDATA <= 32'h0;
        end
        @(posedge clk);
        PRDATA <= 32'hCAFE_F00D; 
        PREADY <= 1'b1;          

        @(posedge clk);
        PREADY <= 1'b0;
        PRDATA <= 32'h0;
        addr   <= 32'h0;
        
  
        #40;
        
        $finish;
    end
    initial begin
        $monitor("Time=%0t | State: PSEL=%b PENABLE=%b | PADDR=%h PWDATA=%h PWRITE=%b | PREADY=%b RDATA=%h EPREADY=%b", 
                 $time, PSEL, PENABLE, PADDR, PWDATA, PWRITE, PREADY, rdata, EPREADY);
    end
    initial begin
        $dumpfile("apb_master.vcd");
        $dumpvars(0, tb_apb_master);
    end

endmodule
