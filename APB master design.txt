module apb_master (

    input clk,
    input reset_n,
   
    input [31:0]addr,       // E
    input [31:0]wdata,      // E
    input EWRITE,     // 1 = Write, 0 = Read
    input EVALID,     // E
    output reg [31:0] rdata,      // E
    output reg EPREADY,    // E
    
    input [31:0]PRDATA,
    input PREADY,
    input PSLVERR,
    output reg [31:0]PADDR,
    output reg [31:0]PWDATA,
    output reg PWRITE,
    output reg PSEL,
    output reg PENABLE);

    parameter IDLE   = 2'b00;
    parameter SETUP  = 2'b01;
    parameter ACCESS = 2'b10;
    reg [1:0] state;
    reg [1:0] next_state;
    
    reg [31:0] addr_reg;
    reg [31:0] wdata_reg;
    reg write_reg;
    
    always @(posedge clk or negedge reset_n)
    begin
        if (!reset_n)
            state <= IDLE;
        else
            state <= next_state;
    end
  
    always @(posedge clk or negedge reset_n)
    begin
        if (!reset_n)
        begin
            addr_reg  <= 32'b0;
            wdata_reg <= 32'b0;
            write_reg <= 1'b0;
        end
        else if (state == IDLE && EVALID)
        begin
            addr_reg  <= addr;
            wdata_reg <= wdata;
            write_reg <= EWRITE;
        end
    end
   
    always @(*)
    begin
        case (state)

            IDLE:
            begin
                if (EVALID)
                    next_state = SETUP;
                else
                    next_state = IDLE;
            end

            SETUP:
            begin
                next_state = ACCESS;
            end

            ACCESS:
            begin
                if (PREADY)
                    next_state = IDLE;
                else
                    next_state = ACCESS;
            end

            default:
            begin
                next_state = IDLE;
            end

        endcase
    end
  
    always @(*)
    begin
        
        PADDR   = 32'b0;
        PWDATA  = 32'b0;
        PWRITE  = 1'b0;
        PSEL    = 1'b0;
        PENABLE = 1'b0;
        rdata   = 32'b0;
        EPREADY = 1'b0;

        case (state)
          IDLE:
            begin
                PSEL    = 1'b0;
                PENABLE = 1'b0;
            end
          SETUP:
            begin
                PADDR   = addr_reg;
                PWDATA  = wdata_reg;
                PWRITE  = write_reg;

                PSEL    = 1'b1;
                PENABLE = 1'b0;
            end
          ACCESS:
            begin
                PADDR   = addr_reg;
                PWDATA  = wdata_reg;
                PWRITE  = write_reg;
                PSEL    = 1'b1;
                PENABLE = 1'b1;
           
                if (PREADY)
                begin
                    EPREADY = 1'b1;
              
                    if (!write_reg)
                    begin
                        rdata = PRDATA;
                    end
                end
            end
            default:
            begin
                PSEL    = 1'b0;
                PENABLE = 1'b0;
            end
        endcase
    end
endmodule
